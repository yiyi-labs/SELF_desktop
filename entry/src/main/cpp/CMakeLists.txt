cmake_minimum_required(VERSION 3.5.0)
project(self_native)
set(CMAKE_CXX_STANDARD 17)
add_library(self_native SHARED napi.cpp RendererSession.cpp ProductScene.cpp ImagePipeline.cpp GltfAssetLoader.cpp RegionSelection.cpp)
target_sources(self_native PRIVATE CameraPipeline.cpp vendor/libfacedetection/facedetectcnn.cpp vendor/libfacedetection/facedetectcnn-model.cpp vendor/libfacedetection/facedetectcnn-data.cpp)
target_sources(self_native PRIVATE CameraCodecs.c CameraMux.c)
set_source_files_properties(CameraCodecs.c PROPERTIES COMPILE_OPTIONS "-O3;-Wno-macro-redefined;-Wno-sign-compare;-Wno-unused-parameter")
set_source_files_properties(CameraPipeline.cpp vendor/libfacedetection/facedetectcnn.cpp vendor/libfacedetection/facedetectcnn-model.cpp vendor/libfacedetection/facedetectcnn-data.cpp PROPERTIES COMPILE_OPTIONS "-O3;-Wno-unused-parameter;-Wno-sign-compare")
if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
    set_source_files_properties(vendor/libfacedetection/facedetectcnn.cpp vendor/libfacedetection/facedetectcnn-model.cpp vendor/libfacedetection/facedetectcnn-data.cpp PROPERTIES COMPILE_DEFINITIONS "_ENABLE_NEON")
endif()
target_compile_options(self_native PRIVATE -Wall -Wextra -Werror=return-type)
target_link_libraries(self_native PUBLIC libace_napi.z.so libace_ndk.z.so libEGL.so libGLESv3.so libnative_window.so libnative_vsync.so libhilog_ndk.z.so libimage_source.so libpixelmap.so libimage_packer.so)
